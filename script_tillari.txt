    <div class="page" id="page-glossary">
      <div class="pg-header">
        <h1>Glossariy</h1>
        <p>Script tillari bo'yicha asosiy atamalar va ta'riflar lug'ati</p>
      </div>
      <div class="card">
        <div class="card-title"><i class="ti ti-vocabulary"></i> Asosiy atamalar</div>
        <div class="g-item"><div class="g-term">Array (Massiv)</div><div class="g-def">Ko'p qiymatlarni tartibli saqlash uchun ishlatiladigan ma'lumotlar tuzilmasi. JavaScript'da: <code>let arr = [1, 2, 3]</code></div></div>
        <div class="g-item"><div class="g-term">Async / Await</div><div class="g-def">Asinxron funksiyalarni sinxron ko'rinishda yozish imkonini beruvchi JavaScript sintaksisi. <code>async function</code> kalit so'zi bilan e'lon qilinadi.</div></div>
        <div class="g-item"><div class="g-term">BOM (Browser Object Model)</div><div class="g-def">Brauzer oynasi bilan ishlash imkonini beruvchi ob'ektlar modeli. Window, Navigator, Screen, History ob'ektlarini o'z ichiga oladi.</div></div>
        <div class="g-item"><div class="g-term">Callback</div><div class="g-def">Boshqa funksiya argumenti sifatida uzatiladigan va keyinchalik chaqiriladigan funksiya.</div></div>
        <div class="g-item"><div class="g-term">Closure (Yopilish)</div><div class="g-def">Ichki funksiyaning tashqi funksiya o'zgaruvchilariga kirishini saqlash xususiyati.</div></div>
        <div class="g-item"><div class="g-term">DOM (Document Object Model)</div><div class="g-def">HTML hujjatini ob'ektlar daraxti sifatida ifodalovchi brauzer interfeysli model. JavaScript orqali o'zgartiriladi.</div></div>
        <div class="g-item"><div class="g-term">Event (Voqea)</div><div class="g-def">Foydalanuvchi harakati yoki brauzer tomonidan ishga tushuriladigan signal (klik, tugma bosish va boshqalar).</div></div>
        <div class="g-item"><div class="g-term">Function (Funksiya)</div><div class="g-def">Ma'lum vazifani bajaruvchi, qayta foydalanish mumkin bo'lgan kod bloki.</div></div>
        <div class="g-item"><div class="g-term">Interpreter (Tarjimon)</div><div class="g-def">Kodni kompilyatsiyasiz, qator-qator bajaradigan dastur. Script tillarida keng qo'llaniladi.</div></div>
        <div class="g-item"><div class="g-term">JSON (JavaScript Object Notation)</div><div class="g-def">Ma'lumotlarni almashish uchun keng qo'llaniladigan yengil format. Kalit-qiymat juftliklari asosida ishlaydi.</div></div>
        <div class="g-item"><div class="g-term">Module (Modul)</div><div class="g-def">Alohida fayl yoki kod birligi bo'lib, <code>import/export</code> orqali boshqa fayllarda ishlatiladi.</div></div>
        <div class="g-item"><div class="g-term">PHP (Hypertext Preprocessor)</div><div class="g-def">Server tomonida ishlaydigan umumiy maqsadli script tili. Asosan veb-dasturlashda keng qo'llaniladi.</div></div>
        <div class="g-item"><div class="g-term">Promise</div><div class="g-def">Asinxron amalning kelajakdagi natijasini ifodalovchi JavaScript ob'ekti. Uch holati bor: pending, fulfilled, rejected.</div></div>
        <div class="g-item"><div class="g-term">Prototype (Prototip)</div><div class="g-def">JavaScript'da ob'ektlar meros olish mexanizmi. Har bir ob'ektning __proto__ xossasi mavjud.</div></div>
        <div class="g-item"><div class="g-term">Ruby</div><div class="g-def">Dinamik, ob'ektga yo'naltirilgan va umumiy maqsadli dasturlash tili. Yukihiro Matsumoto tomonidan yaratilgan.</div></div>
        <div class="g-item"><div class="g-term">Scope (Qamrov)</div><div class="g-def">O'zgaruvchi yoki funksiya qaysi kod sohasida ko'rinishi mumkinligini belgilovchi tushuncha.</div></div>
        <div class="g-item"><div class="g-term">Script tili</div><div class="g-def">Odatda interpretatsiya qilinuvchi, tez yoziluvchi va avtomatlashtirish uchun mo'ljallangan dasturlash tili.</div></div>
        <div class="g-item"><div class="g-term">Variable (O'zgaruvchi)</div><div class="g-def">Ma'lumotlarni saqlash uchun ajratilgan xotira maydoni. JS'da <code>var</code>, <code>let</code>, <code>const</code> bilan e'lon qilinadi.</div></div>
      </div>

  function goByKey(key) {
    const page = document.getElementById('page-' + key);
    if (!page) {
      console.warn('Bunday sahifa topilmadi:', 'page-' + key);
      return;
    }

    document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
    page.classList.add('active');

    document.querySelectorAll('.nav-item').forEach(item => item.classList.remove('active'));
    const navItem = document.querySelector('.nav-item[onclick*="' + key + '"]');
    if (navItem) navItem.classList.add('active');

    document.querySelector('.main').scrollTo({ top: 0, behavior: 'smooth' });
  }

  function go(el, key) {
    goByKey(key);
    document.querySelectorAll('.nav-item').forEach(item => item.classList.remove('active'));
    if (el) el.classList.add('active');
  }

  // URL orqali ochish: index.html#lecture, index.html#sillabus va hokazo
  window.addEventListener('load', () => {
    const hash = location.hash.replace('#', '').trim();
    if (hash && document.getElementById('page-' + hash)) {
      goByKey(hash);
    }
  });
